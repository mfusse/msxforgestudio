cmake_minimum_required(VERSION 3.21)
project(MSXForgeStudio VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_library(forge_core STATIC core/engine.c core/rom.c core/msx2_graphics.c)
target_include_directories(forge_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/core")
add_executable(MSXForgeStudio WIN32 qt/main.cpp qt/project_io.cpp qt/project_io.h qt/resources.qrc)
target_link_libraries(MSXForgeStudio PRIVATE Qt6::Widgets forge_core)
target_compile_definitions(MSXForgeStudio PRIVATE QT_NO_KEYWORDS)
if(MSVC)
  target_compile_options(MSXForgeStudio PRIVATE /W4 /utf-8)
  target_compile_options(forge_core PRIVATE /W3)
else()
  target_compile_options(MSXForgeStudio PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS MSXForgeStudio RUNTIME DESTINATION .)
